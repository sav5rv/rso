// ✅ CORREÇÃO: Função de gerar mensagem (já está correta)
function gerarMensagemOutlook() {
    const destinatario = $('#destinatario').val();
    const assunto = $('#assunto').val();
    const mensagem = $('#mensagem').val(); // Texto puro
    
    if (!destinatario || !assunto || !mensagem) {
        alert('Por favor, preencha todos os campos obrigatórios.');
        return;
    }
    
    // Codificar os parâmetros para o mailto
    const assuntoCodificado = encodeURIComponent(assunto);
    const mensagemCodificada = encodeURIComponent(mensagem); // Texto puro codificado
    
    // Criar o link mailto
    const mailtoLink = `mailto:${destinatario}?subject=${assuntoCodificado}&body=${mensagemCodificada}`;
    
    console.log('Link mailto gerado:', mailtoLink);
    
    // Abrir o cliente de e-mail
    window.location.href = mailtoLink;
    
    // Fechar o modal após um tempo
    setTimeout(() => {
        bootstrap.Modal.getInstance(document.getElementById('outlookModal')).hide();
    }, 1000);
}
